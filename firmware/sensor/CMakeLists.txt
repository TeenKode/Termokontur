cmake_minimum_required(VERSION 3.16)
set(VERSION_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../termokontur/__init__.py")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${VERSION_SOURCE}")
file(STRINGS "${VERSION_SOURCE}" VERSION_LINE REGEX "^__version__ = ")
if(NOT VERSION_LINE MATCHES "^__version__ = \"([0-9]+\\.[0-9]+\\.[0-9]+)\"$")
    message(FATAL_ERROR "Invalid Termokontur release version")
endif()
set(PROJECT_VER "${CMAKE_MATCH_1}")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(termokontur-sensor VERSION "${PROJECT_VER}")
